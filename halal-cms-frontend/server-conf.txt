server {
    listen 80;
    root /usr/share/nginx/html;
    index index.html;

    client_max_body_size 50m;

    # Security headers
    add_header X-Frame-Options DENY always;
    add_header X-Content-Type-Options nosniff always;
    add_header Referrer-Policy strict-origin-when-cross-origin always;

    # Gzip compression
    gzip on;
    gzip_types text/plain text/css application/javascript application/json image/svg+xml;
    gzip_min_length 1024;

    # Proxy settings shared across all upstream locations
    proxy_http_version 1.1;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_read_timeout 60s;

    # ── Auth Service ─────────────────────────────────────────────────
    location /api/auth/ {
        proxy_pass http://auth-service:8081/auth/;
    }
    location /api/mgmt/ {
        proxy_pass http://auth-service:8081/mgmt/;
    }
    location /api/users/ {
        proxy_pass http://auth-service:8081/users/;
    }

    # ── Company Service ──────────────────────────────────────────────
    location /api/companies/ {
        proxy_pass http://company-service:8083/companies/;
    }

    # ── Application Service ──────────────────────────────────────────
    location /api/applications/ {
        proxy_pass http://application-service:8082/applications/;
    }
    location /api/dashboard/ {
        proxy_pass http://application-service:8082/dashboard/;
    }
    location /api/system/ {
        proxy_pass http://application-service:8082/system/;
    }

    # ── Inspection Service ───────────────────────────────────────────
    location /api/audits/ {
        proxy_pass http://inspection-service:8084/audits/;
    }
    location /api/audit-plans/ {
        proxy_pass http://inspection-service:8084/audit-plans/;
    }

    # ── Certificate Service ──────────────────────────────────────────
    location /api/certificates/ {
        proxy_pass http://certificate-service:8085/certificates/;
    }

    # ── React SPA (client-side routing) ─────────────────────────────
    location / {
        try_files $uri $uri/ /index.html;
    }
}
